    \begin{enumerate}
        \item $\Omega := \{1, 2, 3, ..., 11, 12\}$ % Ergebnismenge
        \item $\mathfrak{P}(\Omega) := \{\{1\},\{2\}, \{3\}, ..., \{12\}\} \Rightarrow $12 mögliche Ereignisse % Menge der möglichen Ereignisse
        \item $A = \{1, 3, 5, ..., 9, 11\} \Rightarrow$ 6 mögliche Ereignisse $\Rightarrow 6/12$
        \item $B = \{2\} \Rightarrow$ 1 mögliches Ereignis
        \item $C = \{2, 3\} \Rightarrow$ 2 mögliche Ereignisse
        \item $D = \emptyset \Rightarrow$ nicht gewürfelt, kein Ereignis
    \end{enumerate}
    \subsubsection{}
    \begin{enumerate}
        \item $A \cap B \rightarrow B \notin A \land A \notin B \Rightarrow$ A und B unvereinbar
        \item $B \in C \Rightarrow$ B und C vereinbar
        \item $D := \emptyset \rightarrow D \subseteq A : A \setminus D \subseteq A \Rightarrow A \setminus A = \emptyset \Leftrightarrow \emptyset \subseteq A \Rightarrow$ D und A vereinbar
        \item C und D vereinbar, s.o.
    \end{enumerate}
